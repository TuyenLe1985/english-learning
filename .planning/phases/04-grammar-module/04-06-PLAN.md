// Forwards the raw JWE session cookie string as a Bearer header to NestJS.

```typescript
const session = await auth();
if (!session) { /* 401 / redirect */ }
const reqHeaders = await headers();          // from "next/headers"
const cookieHeader = reqHeaders.get("cookie") ?? "";
const res = await fetchWithAuth(cookieHeader, `${API_URL}/api/grammar/areas`);
```

```typescript

```
