  <behavior>
    GREEN targets from reading.service.spec.ts RED scaffolds:

    getPassages(userId, query): calls prisma.readingPassage.findMany with isPublished:true + filter spread; includes _count.questions and bookmarks where userId; returns { passages, total, page, limit, totalPages }

    getPassageById(userId, id): calls prisma.readingPassage.findUnique with include of questions (ordered by sortOrder), highlights (where userId), note (findFirst where {userId, passageId}), progress (findUnique where userId_passageId); throws NotFoundException if not found

    completeSession(userId, dto): calls prisma.readingProgress.upsert with where: { userId_passageId: { userId, passageId } }; create+update both set score, accuracy, readingTimeSec, completedAt, lastViewedAt

    createHighlight(userId, dto): calls prisma.highlight.create with { userId, passageId, startOffset, endOffset, text }; returns the created highlight including id

    deleteHighlight(userId, id): fetches highlight first, verifies highlight.userId === userId (IDOR protection — throw ForbiddenException if mismatch), then calls prisma.highlight.delete

    upsertNote(userId, dto): calls prisma.note.upsert with where: { userId_passageId: { userId, passageId } }; create sets userId+passageId+content; update sets content

    toggleBookmark(userId, dto): checks prisma.bookmark.findUnique where userId_passageId; if exists → delete and return {bookmarked:false}; if not → create and return {bookmarked:true}
  </behavior>
