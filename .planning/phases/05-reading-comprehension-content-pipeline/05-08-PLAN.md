Sentence extraction (D-15):
```typescript
function extractSentence(fullText: string, targetWord: string): string {
  const sentences = fullText.split(/[.!?](?:\s|$)/);
  const lower = targetWord.toLowerCase();
  return sentences.find(s => s.toLowerCase().includes(lower)) ?? sentences[0] ?? '';
}
```
